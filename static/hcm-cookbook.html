Query #1 — Active Headcount (zero duplicates)
-- Headcount: one row per active employee, guaranteed.
-- _M table + EFFECTIVE_LATEST_CHANGE handles future-dated rows automatically.
SELECT
  p.full_name,
  a.assignment_number,
  a.department_id,
  a.location_id,
  a.job_id,
  a.grade_id
FROM  per_all_assignments_m  a
JOIN  per_all_people_f       p  ON p.person_id = a.person_id
                                AND p.effective_start_date <= SYSDATE
                                AND p.effective_end_date   >= SYSDATE
WHERE a.effective_latest_change  = 'Y'        -- one row per assignment
  AND a.assignment_type          = 'E'        -- employees only (not CWK)
  AND a.assignment_status_type    = 'ACTIVE_ASSIGN'
  AND a.primary_flag             = 'Y'        -- primary assignment only
ORDER BY p.full_name;
